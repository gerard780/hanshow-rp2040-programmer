cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD waveshare_rp2040_zero CACHE STRING "Target board")
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(hanshow_pio_bridge C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
option(ENABLE_TAG_RESET "Allow CDC0 RTS to drive confirmed tag reset on GPIO2" OFF)
add_executable(hanshow_pio_bridge src/main.c src/usb_descriptors.c)
target_include_directories(hanshow_pio_bridge PRIVATE src)
target_compile_definitions(hanshow_pio_bridge PRIVATE ENABLE_TAG_RESET=$<BOOL:${ENABLE_TAG_RESET}>)
target_compile_definitions(hanshow_pio_bridge PRIVATE PICO_CXX_DISABLE_ALLOCATION_OVERRIDES=1)
set_target_properties(hanshow_pio_bridge PROPERTIES LINKER_LANGUAGE C)
target_compile_options(hanshow_pio_bridge PRIVATE -Wall -Wextra)
target_link_libraries(hanshow_pio_bridge PRIVATE pico_stdlib pico_multicore
    pico_unique_id hardware_pio hardware_uart hardware_dma tinyusb_device tinyusb_board)
pico_generate_pio_header(hanshow_pio_bridge ${CMAKE_CURRENT_LIST_DIR}/src/serial.pio)
pico_enable_stdio_uart(hanshow_pio_bridge 0)
pico_enable_stdio_usb(hanshow_pio_bridge 0)
pico_add_extra_outputs(hanshow_pio_bridge)
